📖 Chapter 1 — Introduction to Cybersecurity & VAPT

1.1 What is Cybersecurity?

Cybersecurity is the practice of protecting systems, networks, programs, devices and data from digital attacks, unauthorized access, damage or theft. It encompasses everything from end-user awareness to sophisticated cryptographic protocols.

1.6 Real-World Scenario: Why VAPT Saved a Company $4M

📍 Scenario — FinTech Startup "PayX"

Background: PayX, a Bangalore-based fintech startup with 2M users, was weeks from a $40M Series B funding round. Their investors demanded a VAPT before the deal.

The Test: Grey-box web app pentest + API testing over 10 days.

Critical Findings:

  1. CVSS 9.8 — IDOR in transaction API allowed transferring funds between any accounts.
  2. CVSS 9.1 — Hardcoded AWS access keys in mobile app APK → S3 bucket with 8M user KYC documents.
  3. CVSS 8.6 — Outdated Apache Struts (CVE-2017-5638 — the Equifax bug!).

Impact: PayX fixed all criticals in 21 days. The investor technical due-diligence passed. The Series B closed. Without VAPT, the keys alone could have triggered GDPR/RBI fines >$50M and total reputational collapse.

Lesson: A 10-day VAPT costing $15K saved a $40M raise and prevented a potential breach worth >$50M. ROI: 3,000x.

5.1 Python for Pentesters

5.1.1 Port Scanner (your first tool)

#!/usr/bin/env python3
import socket
import argparse
from concurrent.futures import ThreadPoolExecutor

def scan_port(host, port, timeout=1):
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(timeout)
        result = s.connect_ex((host, port))
        s.close()
        return port if result == 0 else None
    except socket.gaierror:
        return None

def main():
    parser = argparse.ArgumentParser(description='Fast port scanner')
    parser.add_argument('host', help='Target host or IP')
    parser.add_argument('-p', '--ports', default='1-1024',
                        help='Port range (e.g. 1-1024 or 80,443,8080)')
    parser.add_argument('-t', '--threads', type=int, default=100)
    args = parser.parse_args()

    ports = []
    if '-' in args.ports:
        start, end = map(int, args.ports.split('-'))
        ports = range(start, end+1)
    else:
        ports = [int(p) for p in args.ports.split(',')]

    print(f'[*] Scanning {args.host} ports {args.ports}')
    with ThreadPoolExecutor(max_workers=args.threads) as ex:
        results = ex.map(lambda p: scan_port(args.host, p), ports)

    for port in results:
        if port:
            try:
                banner = socket.getservbyport(port)
                print(f'[+] {port}/tcp OPEN ({banner})')
            except:

#!/usr/bin/env python3
import requests
import argparse
from concurrent.futures import ThreadPoolExecutor

def check_dir(url, word):
    try:
        r = requests.get(f'{url}/{word}', timeout=3, allow_redirects=False)
        return (r.status_code, word, len(r.content))
    except:
        return None

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('url', help='Target URL (http://target.com)')
    parser.add_argument('-w', '--wordlist', default='/usr/share/seclists/Discovery/Web-Content/raft-medium-directories.txt')
    parser.add_argument('-t', '--threads', type=int, default=20)
    parser.add_argument('--status', default='200,301,302,403',
                        help='Filter by status codes (comma-separated)')
    parser.add_argument('--extensions', default='', help='e.g. .php,.html,.bak')
    args = parser.parse_args()

    wanted = set(map(int, args.status.split(',')))

    with open(args.wordlist) as f:
        words = [line.strip() for line in f if line.strip()]

    print(f'[*] {len(words)} words, threads={args.threads}')
    with ThreadPoolExecutor(max_workers=args.threads) as ex:
        for status, word, length in ex.map(
            lambda w: check_dir(args.url, w), words):
            if status in wanted:
                print(f'  [{status}] /{word}  ({length} bytes)')

if __name__ == '__main__':
    main()

#!/usr/bin/env python3
import dns.resolver
import requests

# Fingerprints of dangling CNAMEs orphaned services
TAKEOVER_SERVICES = {
    's3.amazonaws.com': 'claim S3 bucket',
    'herokuapp.com': 'register Heroku app',
    'azurewebsites.net': 'claim Azure web app',
    'cloudfront.net': 'claim CloudFront distribution',
    'github.io': 'register GitHub Pages repo',
    'pantheon.io': 'claim Pantheon site',
    'fastly.net': 'claim Fastly service',
    'elasticbeanstalk.com': 'claim Elastic Beanstalk env',
}

def check_subdomain(sub):
    try:
        answers = dns.resolver.resolve(sub, 'CNAME')
        for rdata in answers:
            cname = str(rdata.target).rstrip('.')
            for service, action in TAKEOVER_SERVICES.items():
                if service in cname:
                    try:
                        r = requests.get(f'http://{sub}', timeout=5)
                        if r.status_code in [404, 500] or 'NoSuchBucket' in r.text:
                            return f'[🚨 VULN] {sub} → {cname} → {action}'
                    except: pass
    except dns.resolver.NoAnswer:
        pass
    except: pass
    return None

if __name__ == '__main__':
    import sys
    for sub in open(sys.argv[1]).read().splitlines():
        result = check_subdomain(sub.strip())
        if result: print(result)
